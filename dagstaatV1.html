-- ============================================================ -- GEPLANDE DUIKOEFENINGEN -- Uit te voeren in het Dagstaat-project (SQL Editor). -- Losse, kleine planningstool specifiek voor het duikteam (altijd Arnhem -- Zuid), los van het algemene, ploeg-brede oefenrooster. Bij het markeren -- als 'uitgevoerd' worden automatisch de losse duik_logregels per -- deelnemer aangemaakt (zie code in dagplanning-brandweer.html). -- ============================================================ create table if not exists public.duik_oefeningen ( id text primary key, datum date not null, post_id text not null default 'post2', codes text[] not null default '{}', -- BRW-codes waar dit voor telt (kan combineren) deelnemer_ids text[] not null default '{}', omschrijving text, uitgevoerd boolean not null default false, uitgevoerd_op timestamptz, aangemaakt_door text, aangemaakt_op timestamptz not null default now() ); create index if not exists idx_duik_oefeningen_datum on public.duik_oefeningen (datum); alter table public.duik_oefeningen enable row level security; drop policy if exists "ingelogd mag duik_oefeningen lezen" on public.duik_oefeningen; create policy "ingelogd mag duik_oefeningen lezen" on public.duik_oefeningen for select using (auth.role()='authenticated'); drop policy if exists "ingelogd mag duik_oefeningen wijzigen" on public.duik_oefeningen; create policy "ingelogd mag duik_oefeningen wijzigen" on public.duik_oefeningen for all using (auth.role()='authenticated') with check (auth.role()='authenticated'); -- Publiek leesbaar, maar ALLEEN de velden die het kazernedashboard nodig -- heeft (geen deelnemerslijst) -- via een view, zodat de brede tabel zelf -- besloten blijft. create or replace view public.dashboard_duikoefeningen as select id, datum, codes, omschrijving from public.duik_oefeningen where uitgevoerd = false and datum >= current_date order by datum; grant select on public.dashboard_duikoefeningen to anon, authenticated; select count(*) as bestaande_rijen from public.duik_oefeningen;
